// src/scenes/MapScene.js
import * as THREE from 'three';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls.js';
import { BaseScene } from './BaseScene.js';
import { LEVEL_CONFIGS } from '../config/LevelConfigs.js';

export class MapScene extends BaseScene {
 constructor() {
   super();

   this.gltfLoader = new GLTFLoader();
   this.mapSceneMesh = null;
   this.boatMesh = null;
   this.mixer = null;
   this.animations = [];
   this.activeAction = null;
   this.controls = null;

   this.initScene();
 }

 async initScene() {
   // 1. Apply spin scene lighting configuration
   this.applyLightingConfig('spin');

   // 2. Set explicit camera position
   this.setupCamera();

   // 3. Load GLTF Models for map scene and boat
   await Promise.all([
     this.loadMapEnvironment('./public/models/map_scene.glb'),
     this.loadBoatModel('./public/models/map_boat.glb')
   ]);

   // Fallback: Render procedural islands if map GLTF is missing
   if (!this.mapSceneMesh) {
     this.buildProceduralFallbackMap();
   }
 }

 /**
  * Sets up camera with explicit initial coordinates
  */
 setupCamera(domElement = null) {
   if (!this.camera) {
     this.camera = new THREE.PerspectiveCamera(
       45,
       window.innerWidth / window.innerHeight,
       0.1,
       1000
     );
   }

   // 🎯 1. EXPLICIT CAMERA POSITION (Replace with your logged values)
   this.camera.position.set(0.00, 2.00, 10.00);

   // 🎯 2. EXPLICIT CAMERA ROTATION (Optional if not using OrbitControls)
   // this.camera.rotation.set(-0.20, 0.00, 0.00);

   const canvas = domElement || document.querySelector('canvas');
   if (canvas) {
     this.controls = new OrbitControls(this.camera, canvas);
     this.controls.enableDamping = true;
     this.controls.dampingFactor = 0.05;

     // 🎯 3. EXPLICIT ORBIT TARGET POINT (The point the camera revolves around)
     this.controls.target.set(0.00, 0.00, 0.00);
     
     // Update controls to sync with initial camera setup
     this.controls.update();
   } else {
     // Fallback lookAt if OrbitControls is unavailable
     this.camera.lookAt(0, 0, 0);
     console.warn('Canvas DOM element not found for OrbitControls.');
   }
 }

 /**
  * Updates OrbitControls smooth damping frame-by-frame
  */
 update() {
   if (this.controls) {
     this.controls.update();
   }
 }

 /**
  * Loads the 3-Island Environment GLTF Model
  */
 async loadMapEnvironment(path) {
   try {
     const gltf = await new Promise((resolve, reject) => {
       this.gltfLoader.load(path, resolve, undefined, reject);
     });

     this.mapSceneMesh = gltf.scene;
     this.mapSceneMesh.traverse((child) => {
       if (child.isMesh) {
         child.castShadow = true;
         child.receiveShadow = true;
       }
     });
     this.group.add(this.mapSceneMesh);
   } catch (err) {
     console.warn(`map_scene.glb not found at ${path}. Using procedural fallback map.`, err);
   }
 }

 /**
  * Loads the Boat GLTF Model & extracts NLA keyframe animation tracks
  */
 async loadBoatModel(path) {
   try {
     const gltf = await new Promise((resolve, reject) => {
       this.gltfLoader.load(path, resolve, undefined, reject);
     });

     this.boatMesh = gltf.scene;
     this.boatMesh.traverse((child) => {
       if (child.isMesh) {
         child.castShadow = true;
         child.receiveShadow = true;
       }
     });
     this.group.add(this.boatMesh);

     // Setup Animation Mixer for NLA Tracks (level1to2, level2to3)
     if (gltf.animations && gltf.animations.length > 0) {
       this.mixer = new THREE.AnimationMixer(this.boatMesh);
       this.animations = gltf.animations;
     }
   } catch (err) {
     console.warn(`map_boat.glb not found at ${path}. Using procedural fallback ship.`, err);
     this.buildProceduralShip();
   }
 }

 /**
  * Animates travel between islands using Blender NLA tracks or lerp fallback
  */
 animateTravel(fromLevel, toLevel, onComplete) {
   const safeFrom = LEVEL_CONFIGS[fromLevel] ? fromLevel : 1;
   const safeTo = LEVEL_CONFIGS[toLevel] ? toLevel : Math.min(safeFrom + 1, Object.keys(LEVEL_CONFIGS).length);

   // Matches Blender NLA Action names: "level1to2", "level2to3"
   const clipName = `level${safeFrom}to${safeTo}`;
   const clip = this.animations.length ? THREE.AnimationClip.findByName(this.animations, clipName) : null;

   if (this.mixer && clip) {
     // --- NLA KEYFRAME ANIMATION PLAYBACK ---
     if (this.activeAction) {
       this.activeAction.stop();
     }

     this.activeAction = this.mixer.clipAction(clip);

     this.isSpinning = false;
     return;

   if (diff > 0) diff -= Math.PI * 2;

   const finalRotation = currentRot + diff - extraSpins;

   const duration = 4000;
   const startTime = performance.now();
   const startRotation = currentRot;

Response: Here is the updated AudioManager.js.
A GainNode (this.masterGain) has been added between the audio sources and the output destination, set to 0.5 (50% volume). A setVolume() helper function was also added in case you ever want to adjust it dynamically at runtime.
// src/core/AudioManager.js
import { eventBus } from './EventBus.js';

export class AudioManager {
 constructor() {
   this.ctx = null;
   this.masterGain = null; // 🔊 Master Gain Node for global volume
   this.buffers = new Map();
   this.isUnlocked = false;

   this.soundManifest = {
     AU_Spin: './public/sounds/AU_Spin.mp3',
     AU_Spin_Click: './public/sounds/AU_Spin_Click.mp3',
     AU_Popping: './public/sounds/AU_Popping.mp3',
     AU_Cheer: './public/sounds/AU_Cheer.mp3',
     AU_Box_Open: './public/sounds/AU_Box_Open.mp3',
     AU_Click: './public/sounds/AU_Click.mp3',
     AU_Coin: './public/sounds/AU_Coin.mp3',
   };

   this.init();
 }

 init() {
   // 1. Listen for global sound emission events
   eventBus.on('PLAY_SOUND', (soundKey) => this.playSound(soundKey));

   // 2. Setup user interaction listener to unlock AudioContext on first click/touch
   const unlockAudio = () => {
     if (!this.ctx) {
       const AudioContextClass = window.AudioContext || window.webkitAudioContext;
       if (AudioContextClass) {
         this.ctx = new AudioContextClass();

         // 🔊 Create Master Gain Node & set volume to 50% (0.5)
         this.masterGain = this.ctx.createGain();
         this.masterGain.gain.value = 0.5;
         this.masterGain.connect(this.ctx.destination);

         this.preloadSounds();
       }
     }

     if (this.ctx && this.ctx.state === 'suspended') {
       this.ctx.resume();
     }

     this.isUnlocked = true;
     window.removeEventListener('pointerdown', unlockAudio);
     window.removeEventListener('keydown', unlockAudio);
   };

   window.addEventListener('pointerdown', unlockAudio);
   window.addEventListener('keydown', unlockAudio);
 }

 /**
  * Preloads and decodes sound files into memory buffers
  */
 async preloadSounds() {
   for (const [key, path] of Object.entries(this.soundManifest)) {
     try {
       const response = await fetch(path);
       const arrayBuffer = await response.arrayBuffer();
       const decodedData = await this.ctx.decodeAudioData(arrayBuffer);
       this.buffers.set(key, decodedData);
     } catch (err) {
       console.warn(`[AudioManager] Failed to load sound '${key}' from ${path}`, err);
     }
   }
 }

 /**
  * Plays a preloaded sound buffer routed through the Master Gain
  */
 playSound(key) {
   if (!this.ctx || this.ctx.state !== 'running') return;

   const buffer = this.buffers.get(key);
   if (!buffer) return;

   const source = this.ctx.createBufferSource();
   source.buffer = buffer;

   // 🔊 Route audio through masterGain node instead of directly to destination
   if (this.masterGain) {
     source.connect(this.masterGain);
   } else {
     source.connect(this.ctx.destination);
   }

   source.start(0);
 }

 /**
  * Dynamically adjust global volume on the fly (0.0 to 1.0)
  */
 setVolume(volume = 0.5) {
   if (this.masterGain) {
     this.masterGain.gain.value = Math.max(0, Math.min(1, volume));
   }
 }
}

User prompt: I am having problems controlling directional light, skylight, sky color, and background color in all of my scenes, can you make me understand which settings controls what in my level config and IslandScene? I want to start with level1. in this scene there are no skylight and no shadows from directional light
Response: Here is a clear breakdown of how your lighting, sky, and background configuration works, why LIGHTING_CONFIGS is currently being ignored in IslandScene, and why shadows aren't displaying in Level 1.
1. What Controls What?
Currently, there is a disconnect between your config files and IslandScene.js.


Configuration / File
	What It Controls
	Current Status in IslandScene
	LEVEL_CONFIGS
	Gameplay data per island (Camera position, map Y-offset, building costs, ground/sea color values for custom shaders).
	Active for game logic and procedural material colors. Does not control lighting.
	LIGHTING_CONFIGS
	Central lighting profiles (backgroundColor, skylight, directional intensity/color/shadows) per scene (island, spin, box).
	❌ Ignored in IslandScene! SpinScene calls applyLightingConfig('spin'), but IslandScene never calls applyLightingConfig('island').
	IslandScene.js
	Creates and manages the 3D scene objects.
	⚠️ Hardcodes its own lights and overrides backgroundColor by adding a physical Sky shader box object.
	2. Why Skylight & Background Color Aren't Working
In IslandScene.js, your constructor manually runs this.setupSkylight() and this.setupSkybox():
// ❌ Hardcoded inside setupSkylight() in IslandScene.js:
const skylight = new THREE.HemisphereLight(0x38bdf8, 0xfcd34d, 0.95);

1. Skylight Override: IslandScene creates a hardcoded HemisphereLight(0x38bdf8, 0xfcd34d, 0.95) instead of reading skyColor, groundColor, and intensity from LIGHTING_CONFIGS.island.skylight.
2. Background Color Override: setupSkybox() adds a physical procedural Sky mesh object scaled to 450000 units. This physical sky dome visually blocks and replaces any backgroundColor set on scene.background.
3. Why Directional Light Shadows Are Missing in Level 1
Even though sunLight.castShadow = true is set in setupSkylight(), shadows usually disappear in Three.js for three specific reasons:
Reason A: Global Renderer ShadowMap is Disabled
In your main setup file (where THREE.WebGLRenderer is created), shadow mapping must be explicitly enabled:
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap; // Optional for soft shadows

If renderer.shadowMap.enabled is false, no shadows will draw anywhere in the scene regardless of light settings.
Reason B: Sun Light Position Moves Outside the Shadow Frustum
In setupSkybox(), you recalculate the light position based on spherical coordinates:
this.sunLight.position.copy(sunPosition).multiplyScalar(50);

This places sunLight at approximately (13.5, 44.1, -19.2). However, your shadow camera bounding box is set to 15 units:
const shadowBoxSize = 15;
sunLight.shadow.camera.left = -shadowBoxSize;
sunLight.shadow.camera.right = shadowBoxSize;
...

If the island mesh or buildings lie outside this 30×30 unit orthographic shadow box from the sun's perspective, shadows are clipped and will not render.
Reason C: GLTF Model Materials (Unlit Materials)
If island_1.glb was exported from Blender with Unlit / Basic materials, Three.js imports them as MeshBasicMaterial. MeshBasicMaterial does not receive shadows or react to lights.
🛠️ The Fix: Clean Integration for IslandScene.js
Here is the updated IslandScene.js that connects to LIGHTING_CONFIGS.island, centers the shadow camera over the island, and properly enables lighting and shadows:
import * as THREE from 'three';
import { BaseScene } from './BaseScene.js';
import { LEVEL_CONFIGS, LIGHTING_CONFIGS } from '../config/LevelConfigs.js';
import { assetLoader } from '../loaders/AssetLoader.js';
import { Sky } from 'three/addons/objects/Sky.js';

export class IslandScene extends BaseScene {
 constructor() {
   super();

   this.buildingMeshes = {
     dock: null,
     lighthouse: null,
     boat: null
   };

   this.islandModel = null;
   
   // 1. Apply central config settings
   this.initLighting();
   
   // 2. Load base island terrain
   this.loadIslandModel();
 }

 initLighting() {
   const config = LIGHTING_CONFIGS.island;

   // 💡 Skylight (Hemisphere Light) from Config
   const skylightConfig = config.skylight;
   this.skylight = new THREE.HemisphereLight(
     skylightConfig.skyColor,
     skylightConfig.groundColor,
     skylightConfig.intensity
   );
   this.skylight.position.set(0, 50, 0);
   this.group.add(this.skylight);

   // ☀️ Directional Light (Sun Light) from Config
   const dirConfig = config.directional;
   this.sunLight = new THREE.DirectionalLight(dirConfig.color, dirConfig.intensity);
   this.sunLight.position.set(...dirConfig.position);
   this.sunLight.castShadow = dirConfig.castShadow;

   // 🎯 Shadow Camera Setup
   this.sunLight.shadow.mapSize.width = 2048;
   this.sunLight.shadow.mapSize.height = 2048;
   this.sunLight.shadow.camera.near = 0.5;
   this.sunLight.shadow.camera.far = 150;

   // Expand frustum size to cover island model completely
   const shadowBoxSize = 25; 
   this.sunLight.shadow.camera.left = -shadowBoxSize;
   this.sunLight.shadow.camera.right = shadowBoxSize;
   this.sunLight.shadow.camera.top = shadowBoxSize;
   this.sunLight.shadow.camera.bottom = -shadowBoxSize;
   this.sunLight.shadow.bias = -0.0005;

   this.group.add(this.sunLight);

   // Optional: Setup Skybox Shader (Comment out if using static scene.background color)
   this.setupSkybox();
 }

 setupSkybox() {
   const sky = new Sky();
   sky.scale.setScalar(450000);
   this.group.add(sky);

   const skyUniforms = sky.material.uniforms;
   skyUniforms['turbidity'].value = 2.2;
   skyUniforms['rayleigh'].value = 0.8;
   skyUniforms['mieCoefficient'].value = 0.003;
   skyUniforms['mieDirectionalG'].value = 0.85;

   const sunPosition = new THREE.Vector3();
   const elevation = 62;
   const azimuth = 145;

   const phi = THREE.MathUtils.degToRad(90 - elevation);
   const theta = THREE.MathUtils.degToRad(azimuth);

   sunPosition.setFromSphericalCoords(1, phi, theta);
   skyUniforms['sunPosition'].value.copy(sunPosition);
 }

 async loadIslandModel() {
   try {
     this.islandModel = await assetLoader.loadGLTF('./public/models/island_1.glb');

     this.islandModel.traverse((child) => {
       if (child.isMesh) {
         child.castShadow = true;
         child.receiveShadow = true;
         
         // Ensure imported materials react to light & shadows
         if (child.material && child.material.isMeshBasicMaterial) {
           child.material = new THREE.MeshStandardMaterial({
             color: child.material.color,
             map: child.material.map || null
           });
         }
       }
     });

     this.islandModel.position.set(0, 0, 0);
     this.group.add(this.islandModel);
     console.log('✅ Island 1 loaded successfully!');
   } catch (err) {
     console.warn('GLTF failed to load, falling back to primitive island:', err);

Why Directional Light Shadows Are Missing in Level 1 Even though sunLight.castShadow = true is set in setupSkylight(), shadows usually disappear in Three.js for three specific reasons: Reason A: Global Renderer ShadowMap is Disabled In your main setup file (where THREE.WebGLRenderer is created), shadow mapping must be explicitly enabled: JavaScript renderer.shadowMap.enabled = true; renderer.shadowMap.type = THREE.PCFSoftShadowMap; // Optional for soft shadows If renderer.shadowMap.enabled is false, no shadows will draw anywhere in the scene regardless of light settings. Reason B: Sun Light Position Moves Outside the Shadow Frustum In setupSkybox(), you recalculate the light position based on spherical coordinates: JavaScript this.sunLight.position.copy(sunPosition).multiplyScalar(50); This places sunLight at approximately (13.5, 44.1, -19.2). However, your shadow camera bounding box is set to 15 units: JavaScript const shadowBoxSize = 15; sunLight.shadow.camera.left = -shadowBoxSize; sunLight.shadow.camera.right = shadowBoxSize; ... If the island mesh or buildings lie outside this $30 \times 30$ unit orthographic shadow box from the sun's perspective, shadows are clipped and will not render. Response: Why Your Shadows Are Washed Out
Yes, high skylight and ambient light intensity will absolutely wash out shadows.
Shadows in Three.js are created by the directional light being blocked. The "darkness" of a shadow depends entirely on how much ambient / skylight is filling in those unlit areas:
   1. Double Lighting Conflict: In App.js, you have an AmbientLight set to 0.8 intensity. At the same time, IslandScene.js creates its own HemisphereLight (skylight) with 0.95 intensity.
   2. Result: Your total ambient scene lighting is 1.75! Because the unlit shadow areas are receiving so much ambient light from both sources, the shadows look flat, faint, or completely invisible.
To fix this, we should lower the global ambient light in App.js (e.g., down to 0.3) so the directional light can create crisp, defined contrast.

 }

   }

   });

   } else if (key === 'spin') {

     this.cameraManager.setBoxCamera();
   }
 }

 setupListeners() {
   window.addEventListener('resize', () => {
     this.cameraManager.updateAspect(window.innerWidth / window.innerHeight);
     this.renderer.setSize(window.innerWidth, window.innerHeight);
   });

       this.dirLight.intensity = 1.5;
     } else {
       this.renderer.shadowMap.enabled = false;
       this.dirLight.castShadow = false;
       this.dirLight.intensity = 1.2;
     }
   });
 }

 start() {
   const animate = () => {
     requestAnimationFrame(animate);

     // 🟢 Update active scene (enables smooth OrbitControls damping in MapScene)
     const activeScene = this.scenes[this.activeSceneKey];
     if (activeScene && typeof activeScene.update === 'function') {
       activeScene.update();
     }

     this.renderer.render(this.scene, this.camera);
   };
   animate();
 }
}

🎮 Three.js Island & Minigame Web App
A modular 3D web game built with Three.js and WebGL, featuring interactive island progression, minigames (Wheel Spin & Box Opening), custom lighting profiles, and event-driven audio management.
🏗️ Architecture & Project Structure
The application follows an object-oriented, scene-managed architecture:
   * App.js: Central application entry point. Handles WebGLRenderer initialization, PCFSoftShadowMap configuration, canvas resizing, global lighting defaults, and active scene swapping via EventBus.
   * CameraManager.js: Manages scene-specific orthographic/perspective camera positioning and smooth controls.
   * AudioManager.js: Central Web Audio API manager. Loads sound manifests into memory buffers and routes playback through a global MasterGainNode (set to 50% volume).
   * GameState.js: Singleton state manager tracking player keys, spins, level progression, and upgrade thresholds.
   * LevelConfigs.js: Central configuration registry storing level progression costs, ground/sea color maps, camera vectors, and standardized LIGHTING_CONFIGS (skylight, directional shadow map properties) across all scenes.
🎬 Scenes Overview
🏝️ 1. IslandScene
   * Renders base island GLTF terrain (island_1.glb) and building attachments (dock, lighthouse, boat).
   * Supports dynamic building instantiation and smooth upgrades using elastic-out scale easing.
   * Configures procedural environment lighting using THREE.Sky and directional sun shadows tuned for orthographic frustums.
🎡 2. SpinScene
   * Renders a 3D wheel minigame with interactive GLTF components (spinner.glb, spin_button.glb) and dynamic 2D canvas text textures mapped onto 3D planes.
   * Handles button-press physics animations and mathematical easing for wheel rotation alignment.
   * Integrates navigation guards preventing soft-locks while spins remain.
📦 3. BoxOpeningScene
   * Renders a 3x3 grid of interactable reward chests (chest.glb) with procedural fallbacks.
   * Plays GLTF keyframe animations via THREE.AnimationMixer upon key usage and reveals reward textures hovering over opened chests.
🗺️ 4. MapScene
   * Renders world progression map environments (map_scene.glb) and ship travel models (map_boat.glb).
   * Executes route travel animations using Blender NLA keyframe tracks (level1to2, level2to3).
